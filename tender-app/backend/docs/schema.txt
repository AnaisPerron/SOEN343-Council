--
-- Name: ensure_matching_profile_role(); Type: FUNCTION; Schema: public; Owner: -
--

CREATE FUNCTION public.ensure_matching_profile_role() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
DECLARE
    expected_role TEXT := TG_ARGV[0];
    actual_role TEXT;
BEGIN
    SELECT role INTO actual_role FROM users WHERE id = NEW.user_id FOR UPDATE;
    IF actual_role IS DISTINCT FROM expected_role THEN
        RAISE EXCEPTION 'User % must have role %, actual role %',
            NEW.user_id, expected_role, actual_role;
    END IF;
    RETURN NEW;
END;
$$;


--
-- Name: ensure_order_item_restaurant(); Type: FUNCTION; Schema: public; Owner: -
--

CREATE FUNCTION public.ensure_order_item_restaurant() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
DECLARE
    ordered_restaurant BIGINT;
    item_restaurant BIGINT;
BEGIN
    SELECT restaurant_id INTO ordered_restaurant
      FROM orders WHERE id = NEW.order_id;

    SELECT m.restaurant_id INTO item_restaurant
      FROM menu_items mi
      JOIN menus m ON m.id = mi.menu_id
      WHERE mi.id = NEW.menu_item_id;

    IF ordered_restaurant IS DISTINCT FROM item_restaurant THEN
        RAISE EXCEPTION 'Menu item % does not belong to restaurant for order %',
            NEW.menu_item_id, NEW.order_id;
    END IF;
    RETURN NEW;
END;
$$;


--
-- Name: prevent_incompatible_user_role_change(); Type: FUNCTION; Schema: public; Owner: -
--

CREATE FUNCTION public.prevent_incompatible_user_role_change() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
BEGIN
    IF NEW.role IS DISTINCT FROM OLD.role THEN
        IF EXISTS (SELECT 1 FROM customer_profiles WHERE user_id = NEW.id)
           OR EXISTS (SELECT 1 FROM courier_profiles WHERE user_id = NEW.id)
           OR EXISTS (SELECT 1 FROM restaurant_staff WHERE user_id = NEW.id) THEN
            RAISE EXCEPTION 'Cannot change role of user % while a profile or staff record exists', NEW.id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$;
